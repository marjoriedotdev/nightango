  • Client methods reach for the global client instead of this (5559, 5618, 5622, 5626, 5735), so a second Client cannot exist, even in a test.
  • Sidebar stores this.client (9904) and still uses the global everywhere.
  • client.connect and client.post exist only after requests() has run (4732, 4738): an ordering constraint no caller can see.

// net/connection.js
function createConnection({ room, openSocket, auth, timeoutMs = 10000 }) {
  state = "closed"; pending = null
  function join() {
    if (pending) return pending           // same promise, always settles
    pending = new Promise((resolve, reject) => {
      state = "connecting"
      socket = openSocket(serverFor(room))
      timer = setTimeout(() => fail(reject, "timeout"), timeoutMs)
      socket.onmessage = x => {
        event = decode(x.data)
        if (event.type == "joined") open(resolve)
        emit("event", event)
      }
    }).finally(() => pending = null)
    return pending
  }
  return { join, leave, get state() { return state }, on }
}

// member/resources.js
function createResourceCache({ fetchText, parse }) {
  inflight = new Map(); cache = new Map()
  return {
    get(username, kind, { reload } = {}) {
      key = kind + ":" + username.toLowerCase()
      if (!reload && cache.has(key)) return Promise.resolve(cache.get(key))
      if (!inflight.has(key))

          .then(text => parse[kind](text))
          .then(rec => (cache.set(key, rec), rec))
          .finally(() => inflight.delete(key)))
      return inflight.get(key)
    }
  }
}
